# Replikayı durdur
$ docker compose stop mysql-replica
 Container yemekhane-mysql-replica-1 Stopping 
 Container yemekhane-mysql-replica-1 Stopped 

# Replika kapalıyken kaynakta bakiye yükle
$ docker compose exec -T mysql-primary mysql --default-character-set=utf8mb4 -uyemekhane_app -pUyg_2026 yemekhane -e "CALL sp_bakiye_yukle(7, 200)"
hareket_id	bakiye
459	1055.00

# Replikayı başlat
$ docker compose start mysql-replica
 Container yemekhane-mysql-replica-1 Starting 
 Container yemekhane-mysql-replica-1 Started 

# Replika aradaki değişikliği yakaladı
$ docker compose exec -T mysql-replica mysql --default-character-set=utf8mb4 -uyemekhane_app -pUyg_2026 yemekhane -e "SELECT id, ad, soyad, bakiye FROM kullanicilar WHERE id = 7"
id	ad	soyad	bakiye
7	Mehmet	Yıldız	1055.00

$ docker compose exec -T mysql-replica mysql -uroot -pyemekhane_kok -e "SHOW REPLICA STATUS\G" | grep -E 'Replica_IO_Running|Replica_SQL_Running:|Seconds_Behind'

          Replica_SQL_Running: Yes
        Seconds_Behind_Source: 0
