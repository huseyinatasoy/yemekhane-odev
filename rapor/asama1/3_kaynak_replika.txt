# Önce: kaynak ve replika aynı bakiyeyi gösteriyor
$ docker compose exec -T mysql-primary mysql --default-character-set=utf8mb4 -uyemekhane_app -pUyg_2026 yemekhane -e "SELECT id, ad, soyad, bakiye FROM kullanicilar WHERE id = 7"

$ docker compose exec -T mysql-replica mysql --default-character-set=utf8mb4 -uyemekhane_app -pUyg_2026 yemekhane -e "SELECT id, ad, soyad, bakiye FROM kullanicilar WHERE id = 7"
